-- Drop existing policies if they exist
DROP POLICY IF EXISTS "Allow anonymous insert for online booking" ON patients;
DROP POLICY IF EXISTS "Allow anonymous select by contact" ON patients;

-- Allow anonymous users to INSERT patients
CREATE POLICY "Allow anonymous insert for online booking"
ON patients
FOR INSERT
TO anon
WITH CHECK (true);

-- Allow anonymous users to SELECT patients
CREATE POLICY "Allow anonymous select by contact"
ON patients
FOR SELECT
TO anon
USING (true);
